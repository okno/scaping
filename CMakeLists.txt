cmake_minimum_required(VERSION 3.24)
project(scaping VERSION 1.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "SCAPING requires Windows x64 and MSVC (Visual Studio 2022).")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Select the x64 platform (-A x64).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8;/EHsc;/Zc:__cplusplus;/sdl>" "$<$<AND:$<COMPILE_LANGUAGE:CXX>,$<CONFIG:Release>>:/guard:cf>")
add_link_options($<$<CONFIG:Release>:/guard:cf> $<$<CONFIG:Release>:/DEBUG:NONE> /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
add_library(scaping_core STATIC src/core.cpp)
target_include_directories(scaping_core PUBLIC include)
target_link_libraries(scaping_core PUBLIC xmllite shlwapi shell32 ole32)
add_library(scaping_network STATIC src/network.cpp src/worker.cpp)
target_include_directories(scaping_network PUBLIC include PRIVATE src)
target_link_libraries(scaping_network PUBLIC scaping_core ws2_32 iphlpapi advapi32 shell32 user32 wintrust crypt32 bcrypt ole32 uuid)
add_executable(scaping WIN32 src/main.cpp src/ui.cpp resources/scaping.rc)
target_include_directories(scaping PRIVATE include resources)
target_link_libraries(scaping PRIVATE scaping_network comctl32 comdlg32 user32 gdi32 advapi32)
target_link_options(scaping PRIVATE /MANIFEST:NO)
include(CTest)
if(BUILD_TESTING)
  add_executable(scaping_core_tests tests/core_tests.cpp)
  target_link_libraries(scaping_core_tests PRIVATE scaping_core)
  target_compile_definitions(scaping_core_tests PRIVATE SCAPING_FIXTURES_DIR=L"${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME core COMMAND scaping_core_tests)
  add_test(NAME core_english COMMAND scaping_core_tests --language=en)
  set_tests_properties(core core_english PROPERTIES TIMEOUT 60)
  add_executable(scaping_network_tests tests/network_tests.cpp)
  target_link_libraries(scaping_network_tests PRIVATE scaping_network)
  target_include_directories(scaping_network_tests PRIVATE src)
  add_test(NAME network_loopback COMMAND scaping_network_tests)
  add_test(NAME network_loopback_english COMMAND scaping_network_tests --language=en)
  set_tests_properties(network_loopback network_loopback_english PROPERTIES TIMEOUT 40)
endif()
install(TARGETS scaping RUNTIME DESTINATION .)
